local function ScanRemotes()
	CombatRemotes={}
	CombatRemoteSet={}
	pcall(function()
		for _,v in ipairs(ReplicatedStorage:GetDescendants()) do
			if v:IsA("RemoteEvent") then
				local n=string.lower(v.Name)
				if n:find("combat") or n:find("attack") or n:find("hit") or n:find("damage") or n:find("punch") or n:find("swing") or n:find("melee") or n:find("weapon") or n:find("fight") or n:find("deal") or n:find("strike") then
					table.insert(CombatRemotes,v)
					CombatRemoteSet[v]=true
				end
			end
		end
	end)
end

local homeDrag=false

local homeDragStart,homeStartMouse

HomeBtn.InputBegan:Connect(function(input)
	if input.UserInputType==Enum.UserInputType.MouseButton1
	or input.UserInputType==Enum.UserInputType.Touch then
		homeDrag=true
		homeMoved=false
		homeDragStart=input.Position
		homeStartMouse=HomeBtn.Position
		TweenService:Create(HomeBtn,TweenInfo.new(0.15),{Size=UDim2.fromOffset(66,66)}):Play()
	end
end)
UIS.InputChanged:Connect(function(input)
	if homeDrag and (input.UserInputType==Enum.UserInputType.MouseMovement
	or input.UserInputType==Enum.UserInputType.Touch) then
		local delta=input.Position-homeDragStart
		if math.abs(delta.X)>5 or math.abs(delta.Y)>5 then homeMoved=true end
		if homeMoved then
			HomeBtn.Position=UDim2.new(
				homeStartMouse.X.Scale,homeStartMouse.X.Offset+delta.X,
				homeStartMouse.Y.Scale,homeStartMouse.Y.Offset+delta.Y)
		end
	end
end)
UIS.InputEnded:Connect(function(input)
	if input.UserInputType==Enum.UserInputType.MouseButton1
	or input.UserInputType==Enum.UserInputType.Touch then
		homeDrag=false
		TweenService:Create(HomeBtn,TweenInfo.new(0.15),{Size=UDim2.fromOffset(60,60)}):Play()
	end
end)
HomeBtn.MouseButton1Click:Connect(function()
	if homeMoved then return end
	local now=tick()
	if now-homeLastClick<0.25 then return end
	homeLastClick=now
	MenuVisible=not MenuVisible
	if MenuVisible then
		Main.Visible=true
		Main.Size=UDim2.fromOffset(0,0)
		Main.Position=UDim2.new(0.5,0,0.5,0)
		TweenService:Create(Main,TweenInfo.new(0.35,Enum.EasingStyle.Back,Enum.EasingDirection.Out),{

			Position=UDim2.new(0.5,-BaseW*MenuScale/2,0.5,-BaseH*MenuScale/2)
		}):Play()
	else
		local t=TweenService:Create(Main,TweenInfo.new(0.18,Enum.EasingStyle.Quad,Enum.EasingDirection.In),{
			Size=UDim2.fromOffset(0,0),
			Position=UDim2.new(0.5,0,0.5,0)

				TextColor3=Color3.fromRGB(140,140,140)

		end
	end
end

		end
	end)
	UIS.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch) then
			local relX=input.Position.X-track.AbsolutePosition.X
			local pct=math.clamp(relX/track.AbsoluteSize.X,0,1)
			setValue(minV+pct*(maxV-minV))
		end
	end)
	UIS.InputEnded:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
			dragging=false
		end
	end)
	return f
end

local function FindPlayerByName(part)
	if not part or part=="" then return nil end
	part=string.lower(part)
	for _,p in ipairs(Players:GetPlayers()) do
		if p~=LP then
			if string.find(string.lower(p.Name),part,1,true) then return p end
			if p.DisplayName and string.find(string.lower(p.DisplayName),part,1,true) then return p end
		end
	end
	return nil
end

local function FindNearestPlayer(range)
	local nearest=nil
	local myRoot=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not myRoot then return nil end
	local dist=math.huge
	for _,p in ipairs(Players:GetPlayers()) do
		if ValidTarget(p) then
			local r=p.Character:FindFirstChild("HumanoidRootPart")
			if r then
				local d=(myRoot.Position-r.Position).Magnitude
				if d<dist then dist=d; nearest=p end
			end
		end
	end
	if dist<=(range or ATTACK_RANGE) then return nearest end
	return nil
end

local function HookTool(tool)
	if not tool or not tool:IsA("Tool") then return end
	if HookedTools[tool] then return end
	HookedTools[tool]=true
	pcall(function()
		local oldActivate=tool.Activate
		tool.Activate=function(self,...)
			if SPAMATTACK then
				for i=1,AMPLIFY do
					pcall(function() oldActivate(self,...) end)
				end
				return
			end
			return oldActivate(self,...)
		end
	end)
end

local function HookCharacter(char)
	if not char then return end
	for _,v in ipairs(char:GetChildren()) do
		if v:IsA("Tool") then HookTool(v) end
	end
	char.ChildAdded:Connect(function(v)
		if v:IsA("Tool") then HookTool(v) end
	end)
end

local function InstallHook()
	if HookInstalled then return end
	HookInstalled=true
	pcall(function()
		local mt=getrawmetatable(game)
		local old=mt.__namecall
		setreadonly(mt,false)
		mt.__namecall=newcclosure(function(self,...)
			local method=getnamecallmethod()
			if method=="FireServer" and SPAMATTACK then
				if CombatRemoteSet[self] then
					local args={...}
					for i=1,AMPLIFY do
						pcall(function() old(self,unpack(args)) end)
					end
					return
				end
			end
			return old(self,...)
		end)
		setreadonly(mt,true)
	end)
end

local function AttackOnce(target)
	local tool=EquipTool()
	if tool then HookTool(tool) end

	if target and target.Character then
		local myRoot=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
		local tRoot=target.Character:FindFirstChild("HumanoidRootPart")
		if myRoot and tRoot then
			pcall(function()
				myRoot.CFrame=CFrame.lookAt(myRoot.Position,tRoot.Position)
			end)
		end
	end

	if tool then
		if SPAMATTACK then
			for i=1,AMPLIFY do
				pcall(function() tool:Activate() end)
			end
		else
			pcall(function() tool:Activate() end)
		end
	end

	if SPAMATTACK and #CombatRemotes>0 and target and target.Character then
		local tr=target.Character:FindFirstChild("HumanoidRootPart")
		local th=target.Character:FindFirstChildOfClass("Humanoid")
		local remote=CombatRemotes[math.random(1,#CombatRemotes)]
		if tr then pcall(function() remote:FireServer(tr) end) end
		if th then pcall(function() remote:FireServer(th) end) end
		if target then pcall(function() remote:FireServer("Hit",target) end) end
	end
end

local function SnapStickTarget(target)
	if not target or not target.Character then return end
	local tRoot=target.Character:FindFirstChild("HumanoidRootPart")
	local tHum=target.Character:FindFirstChildOfClass("Humanoid")
	local myRoot=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	local myHum=LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
	if not tRoot or not myRoot or not myHum then return end

	pcall(function()
		tRoot.AssemblyLinearVelocity=Vector3.zero
		tRoot.AssemblyAngularVelocity=Vector3.zero
		if tHum then
			tHum.PlatformStand=true
			tHum.WalkSpeed=0
			tHum.JumpPower=0
		end
	end)

	local myPos=myRoot.Position
	local targetPos=tRoot.Position
	local dir=targetPos-myPos
	if dir.Magnitude<0.1 then
		dir=Vector3.new(0,0,1)
	else
		dir=dir.Unit
	end

	local finalCF
	if ORBIT_MODE=="FOLLOW" then
		local tLook=tRoot.CFrame.LookVector
		finalCF=CFrame.lookAt(targetPos-tLook*BEHIND_DIST,targetPos)
	elseif ORBIT_MODE=="STICK" then
		finalCF=CFrame.new(targetPos+Vector3.new(0,0.05,0),targetPos-dir*2)
	elseif ORBIT_MODE=="BEHIND" then
		local tLook=tRoot.CFrame.LookVector
		finalCF=CFrame.lookAt(targetPos-tLook*BEHIND_DIST,targetPos)
	elseif ORBIT_MODE=="ORBIT" then
		orbitAngle=orbitAngle+0.15
		if orbitAngle>math.pi*2 then orbitAngle=orbitAngle-math.pi*2 end
		local ox=math.cos(orbitAngle)*ORBIT_RADIUS
		local oz=math.sin(orbitAngle)*ORBIT_RADIUS
		finalCF=CFrame.lookAt(targetPos+Vector3.new(ox,0.3,oz),targetPos)
	elseif ORBIT_MODE=="INSIDE" then
		finalCF=CFrame.new(targetPos,targetPos-dir*2)
	end

	myHum.AutoRotate=false
	pcall(function()
		myRoot.CFrame=finalCF
	end)
	myRoot.AssemblyLinearVelocity=Vector3.zero
	myRoot.AssemblyAngularVelocity=Vector3.zero
end

local function FindPromptsNear(pos,radius)
	local list={}
	if not pos then return list end
	for _,v in ipairs(Workspace:GetDescendants()) do
		if v:IsA("ProximityPrompt") then
			pcall(function()
				local p=v.Parent
				if p and p:IsA("BasePart") then
					local d=(p.Position-pos).Magnitude
					if d<=radius then
						table.insert(list,{prompt=v,part=p,dist=d})
					end
				elseif p and p:IsA("Attachment") then
					local pp=p.Parent
					if pp and pp:IsA("BasePart") then
						local d=(pp.Position-pos).Magnitude
						if d<=radius then
							table.insert(list,{prompt=v,part=pp,dist=d})
						end
					end
				end
			end)
		end
	end
	table.sort(list,function(a,b) return a.dist<b.dist end)
	return list
end

local function FindClickDetectorsNear(pos,radius)
	local list={}
	if not pos then return list end
	for _,v in ipairs(Workspace:GetDescendants()) do
		if v:IsA("ClickDetector") then
			pcall(function()
				local p=v.Parent
				if p and p:IsA("BasePart") then
					local d=(p.Position-pos).Magnitude
					if d<=radius then
						table.insert(list,{detector=v,part=p,dist=d})
					end
				end
			end)
		end
	end
	table.sort(list,function(a,b) return a.dist<b.dist end)
	return list
end

local function FarmCycle()
	if not FarmLocation then return end
	TeleportTo(FarmLocation)
	task.wait(FarmReturnDelay)
	local prompts=FindPromptsNear(FarmLocation,FarmRadius)
	local detectors=FindClickDetectorsNear(FarmLocation,FarmRadius)
	if #prompts>0 then
		local target=prompts[1]
		for i=1,FarmClicksPerSpot do
			FirePrompt(target.prompt)
			task.wait(FarmClickDelay)
		end
	end
	if #detectors>0 then
		local target=detectors[1]
		for i=1,FarmClicksPerSpot do
			FireClick(target.detector,target.part)
			task.wait(FarmClickDelay)
		end
	end
	if #prompts==0 and #detectors==0 then
		task.wait(FarmClickDelay*FarmClicksPerSpot)
	end
end

local drag=false
local dragStart,startMouse
LogoFrame.InputBegan:Connect(function(input)
	if input.UserInputType==Enum.UserInputType.MouseButton1
	or input.UserInputType==Enum.UserInputType.Touch then
		drag=true
		dragStart=input.Position
		startMouse=Main.Position
	end
end)
UIS.InputChanged:Connect(function(input)
	if drag and (input.UserInputType==Enum.UserInputType.MouseMovement
	or input.UserInputType==Enum.UserInputType.Touch) then
		local delta=input.Position-dragStart
		Main.Position=UDim2.new(
			startMouse.X.Scale,startMouse.X.Offset+delta.X,
			startMouse.Y.Scale,startMouse.Y.Offset+delta.Y)
	end
end)
UIS.InputEnded:Connect(function(input)
	if input.UserInputType==Enum.UserInputType.MouseButton1
	or input.UserInputType==Enum.UserInputType.Touch then
		drag=false
	end
end)

UIS.InputBegan:Connect(function(input,gpe)
	if gpe then return end
	if input.KeyCode==Enum.KeyCode.RightShift then
		MenuVisible=not MenuVisible
		Main.Visible=MenuVisible
		return
	end
end)

	or input.UserInputType==Enum.UserInputType.Touch
	or input.UserInputType==Enum.UserInputType.MouseButton2 then
		if ChatInput and ChatInput.Text~="" then ShowToast(ChatInput.Text) end
	end
end)

RunService.Heartbeat:Connect(function(dt)
	if ANTISTUN then AntiStun() end
	if DASH then
		local c=LP.Character
		if c then
			local hum=c:FindFirstChildOfClass("Humanoid")
			if hum then
				pcall(function()
					hum.WalkSpeed=BloxFruitSpeed
					hum.JumpPower=BloxFruitJump
				end)
			end
		end
	end

	if LOCK then
		local myChar=LP.Character
		local myRoot=myChar and myChar:FindFirstChild("HumanoidRootPart")
		if myRoot then
			if not ValidTarget(LockedTarget) then
				LockedTarget=FindPlayerByName(TargetInput.Text)
				if not LockedTarget then
					LockedTarget=FindNearestPlayer(ATTACK_RANGE)
				end
				if LockedTarget then
					Status.Text="Che do: "..ORBIT_MODE.." | "..LockedTarget.Name
					Status.TextColor3=Color3.fromRGB(0,255,100)
				end
			end
		end
	else
		if LP.Character then
			local hum=LP.Character:FindFirstChildOfClass("Humanoid")
			if hum then hum.AutoRotate=true end
		end
	end
	if AIMBOT then
		local nearest=FindNearestPlayer(ATTACK_RANGE)
		if nearest then
			local c=LP.Character
			local root=c and c:FindFirstChild("HumanoidRootPart")
			local tr=nearest.Character and nearest.Character:FindFirstChild("HumanoidRootPart")
			if root and tr then
				root.CFrame=CFrame.lookAt(root.Position,tr.Position)
			end
		end
	end
	if TP and not LOCK then
		local nearest=FindNearestPlayer(ATTACK_RANGE)
		if nearest then
			local c=LP.Character
			local root=c and c:FindFirstChild("HumanoidRootPart")
			local tr=nearest.Character and nearest.Character:FindFirstChild("HumanoidRootPart")
			if root and tr then
				root.CFrame=tr.CFrame*CFrame.new(0,0,TPDistance)
			end
		end
	end
	for _,p in ipairs(Players:GetPlayers()) do
		if ValidTarget(p) then
			local c=p.Character
			local root=c:FindFirstChild("HumanoidRootPart")
			if ESP then
				if not c:FindFirstChild("HomelessLifeESP") then
					local h=Instance.new("Highlight")
					h.Name="HomelessLifeESP"
					h.FillColor=Color3.fromRGB(0,255,100)
					h.OutlineColor=Color3.new(1,1,1)
					h.FillTransparency=0.5
					h.Parent=c
				end
			else
				local h=c:FindFirstChild("HomelessLifeESP")
				if h then h:Destroy() end
			end
			if HITBOX and root then
				root.Size=Vector3.new(HitboxSize,HitboxSize,HitboxSize)
				root.Transparency=0.7
				root.CanCollide=false
			elseif root then
				root.Size=Vector3.new(2,2,1)
				root.Transparency=1
			end
		end
	end
end)
